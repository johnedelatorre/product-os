  function theme(t, keep) {
    if (t === "light" || t === "dark") document.documentElement.setAttribute("data-theme", t); else { t = "auto"; document.documentElement.removeAttribute("data-theme"); }
    Array.prototype.forEach.call(themeButtons, function (b) { b.setAttribute("aria-pressed", String(b.getAttribute("data-theme-set") === t)); });
    if (keep) { try { window.localStorage.setItem(KEY, t); } catch (e) { /* the choice lasts for this visit only */ } }
  }
  Array.prototype.forEach.call(themeButtons, function (b) { b.addEventListener("click", function () { theme(b.getAttribute("data-theme-set"), true); }); });
  try { theme(window.localStorage.getItem(KEY) || "auto", false); } catch (e) { theme("auto", false); }
  setInterval(stamp, 60000);

  function offline(title, text) {
    gotState = gotDocs = true; state = null; docs = [];
    $("emptyH").textContent = title; $("emptyP").textContent = text; $("emptySteps").hidden = true; live(false, "Not connected"); draw();
  }
  function start(db) {
    var dead = function (e) { clearTimeout(timer); if (e && (e.code === "revoked" || e.code === "not_granted" || e.code === "capability_disabled")) offline("This board cannot read its data here", "Open it in Claude, signed in as the person it belongs to."); else live(false, "The data cannot be reached right now"); };
    timer = setTimeout(function () { if (!gotState || !gotDocs) live(false, "Still connecting. The data cannot be reached yet."); }, 10000);
    function seen(snap) { if (!snap.metadata || !snap.metadata.fromCache) { clearTimeout(timer); } }
    db.doc("project/state").onSnapshot(function (snap) { gotState = true; state = snap.exists ? snap.data() : null; seen(snap); stamp(); draw(); }, dead);
    db.doc("project/links").onSnapshot(function (snap) { links = snap.exists ? snap.data() : null; draw(); }, function () { /* the button stays hidden */ });
    db.collection("documents").limit(200).onSnapshot(function (snap) {
      gotDocs = true; seen(snap);
      docs = snap.docs.map(function (d) { var x = d.data() || {}; return { id: d.id, title: x.title, phase: num(x.phase), kind: x.kind, status: x.status, agent: x.agent, url: x.url, published: x.published, tbd: x.tbd, note: x.note, order: num(x.order) }; })
        .filter(function (x) { return str(x.title, 120); }).sort(function (a, b) { return ((a.phase || 10) - (b.phase || 10)) || (a.order - b.order) || (a.id < b.id ? -1 : 1); });
      draw();
    }, dead);
    db.collection("tickets").limit(300).onSnapshot(function (snap) {
      tickets = snap.docs.map(function (d) { var x = d.data() || {}; return { key: x.key || d.id, type: x.type, summary: x.summary, url: x.url, parent: x.parent, points: x.points, order: num(x.order) }; }).sort(function (a, b) { return a.order - b.order; });
      draw();
    }, function () { /* the tickets list stays as it was */ });
  }
  if (window.claude && typeof window.claude.use === "function") {
    window.claude.use("db").then(function (db) { if (db) start(db); else offline("This board cannot read its data here", "Open it in Claude, signed in as the person it belongs to."); }, function () { offline("This board cannot read its data here", "Open it in Claude, signed in as the person it belongs to."); });
  } else offline("Open this board in Claude", "This is the Product OS board. It shows a project that Claude runs in your chat, so it needs to be opened as a Claude artifact.");
})();
